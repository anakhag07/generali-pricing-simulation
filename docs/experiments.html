279def run_experiment(
280    config: ExperimentConfig,

282) -> ExperimentResult:
283    """Run optimization with all enabled estimators; returns traces and final values."""
284    effective_config = _maybe_apply_acceptance_controls(config)

286    effective_config = _maybe_apply_noise_seed(effective_config, resolved_seeds.noise_seed)
287    objective = effective_config.objective
288    enabled_estimators = tuple(effective_config.enabled_estimators)
289
290    data_rng = rng_from_seed(resolved_seeds.data_seed)
291    if effective_config.theta0 is None:
292        theta0_rng = rng_from_seed(resolved_seeds.theta_seed)
293        policy = getattr(objective, "policy", None)
294        policy_input_dim = getattr(objective, "policy_input_dim", None)
295        theta_state_dim = (
296            int(policy_input_dim()) if callable(policy_input_dim) else effective_config.state_dim
297        )
298        theta_initial = random_theta0(theta_state_dim, policy, theta0_rng)
299        # Persist the resolved theta0 so reporters/plots can access it
300        effective_config = replace(effective_config, theta0=theta_initial)

303

305        if hasattr(effective_config.x_fixed, "iloc") and hasattr(effective_config.x_fixed, "columns"):
306            x_all = effective_config.x_fixed.reset_index(drop=True).copy()
307        else:
308            x_all = np.asarray(effective_config.x_fixed, dtype=float)

319    mean_acceptance_fn = getattr(objective, "mean_acceptance", None)

394            x_samples,
395        )
396        traces[estimator_name] = trace

399    test_metrics: dict[str, PolicyEvaluation] = {}
400    for name, estimator_result in results.items():

402        train_metrics[name] = evaluate_policy(eval_objective, estimator_result.theta, x_samples)
403        if x_test is not None:
