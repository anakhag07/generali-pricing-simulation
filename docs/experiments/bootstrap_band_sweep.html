 30    return sp.Poly.from_list([sp.Rational(float(c)) for c in coefficients[::-1]], VARIABLE, domain=sp.QQ)

 33def lcb_below_level(beta, radius_squared, level):
 34    """Certify an LCB upper level on R; this verifies, never selects, actions."""
 35    g = _poly(beta) - sp.Rational(float(level))
 36    if g.is_zero or nonnegative_on_real_line(-g):
 37        return True
 38    s = radius_squared - g**2
 39    if s.is_zero or nonnegative_on_real_line(s):
 40        return True
 41    # Both strict inequalities define an open set. Sample each exact sign cell
 42    # of g*s, not an evaluation grid. Isolating intervals contain one root each.
 43    product = (g*s).sqf_part()
 44    precision = None
 45    for _ in range(100):
 46        intervals = [interval for interval, _ in product.intervals(eps=precision)]
 47        if not intervals:
 48            samples = [sp.Rational(0)]
 49        else:
 50            samples = [intervals[0][0]-1, intervals[-1][1]+1]
 51            samples += [(left[1]+right[0])/2 for left, right in zip(intervals, intervals[1:])]
 52        # SymPy may return (-1,0) next to an exact root (0,0). Their midpoint
 53        # is then the root, not a sample of the intervening open sign cell.
 54        if all(product.eval(a) != 0 for a in samples):
 55            break
 56        precision = sp.Rational(1, 10**8) if precision is None else precision/100
 57    else:
 58        raise RuntimeError("Could not separate adjacent algebraic sign cells.")
 59    return not any(g.eval(a) > 0 and s.eval(a) < 0 for a in samples)

 62class QuadraticLCBObjective(Objective):
 63    """Policy-free negative LCB, with analytical gradient and no action bounds."""
 64
 65    def __init__(self, beta, covariance, scale):
 66        self.beta = np.asarray(beta, dtype=float)
 67        self.covariance = np.asarray(covariance, dtype=float)
 68        self.scale = float(scale)
 69
 70    def value(self, theta, x_batch):
 71        a = float(theta[0])
 72        p = np.array([1., a, a*a])
 73        return float(-p @ self.beta + self.scale*np.sqrt(p @ self.covariance @ p))
 74
 75    def grad(self, theta, x_batch):
 76        a = float(theta[0])
 77        p, dp = np.array([1., a, a*a]), np.array([0., 1., 2*a])
 78        return np.array([-dp @ self.beta + self.scale*(dp @ self.covariance @ p)/np.sqrt(p @ self.covariance @ p)])

 81def _candidate_lower_value(beta, radius_squared, action):
 82    """Outward-round a candidate value before certifying its objective gap."""
 83    a = sp.Rational(float(action))
 84    q = radius_squared.eval(a)
 85    root_upper = float(np.nextafter(np.sqrt(float(q)), np.inf)) if q else 0.
 86    while sp.Rational(root_upper)**2 < q:
 87        root_upper = float(np.nextafter(root_upper, np.inf))
 88    exact_lower = _poly(beta).eval(a) - sp.Rational(root_upper)
 89    return float(np.nextafter(float(exact_lower), -np.inf))

 92def optimize_lcb(beta, covariance, sigma_hat, critical, config):
 93    """Use repository multistart optimization and certify its all-real gap."""
 94    cert = QuadraticErrorCertificate(covariance, sigma_hat)
 95    radius_squared = cert.q.mul_ground(sp.Rational(float(critical))**2)
 96    leading = sp.Rational(float(beta[2]))
 97    tail_square = radius_squared.nth(4)
 98    # Comparing squares is exact when beta_2 >= 0; negative beta_2 is coercive.
 99    if leading >= 0 and leading**2 >= tail_square:
100        return {"optimization_state": "unbounded" if leading**2 > tail_square else "degenerate_tail",
101                "attempts": [], "action": None, "global_gap_upper": None}
102    objective = QuadraticLCBObjective(beta, covariance, critical*sigma_hat)
103    attempts = []
104    for start in config["starts"]:
105        theta, trace = run_first_order_minimize(
106            np.array([start], dtype=float), np.zeros((1, 1)), objective,
107            t_steps=config["t_steps"], n_grad_samples=1, sigma=0.,
108            algorithm=config["step_rule"], grad_norm_tol=config["gradient_tolerance"],
109            ftol=config["ftol"],
110        )
111        a = float(theta[0])
112        attempts.append({"start": start, "action": a,
113                         "value": float(-objective.value(theta, None)),
114                         "success": bool(trace.optimizer_success),
115                         "status": trace.optimizer_status, "message": trace.optimizer_message,
116                         "gradient_norm": float(np.linalg.norm(objective.grad(theta, None)))})
117    finite = [attempt for attempt in attempts if np.isfinite([attempt["action"], attempt["value"]]).all()]
118    if not finite:
119        return {"optimization_state": "optimizer_failed", "attempts": attempts,
120                "action": None, "global_gap_upper": None}
121    # Select only among actual repository solver outputs, never grid/root actions.
122    best = max(finite, key=lambda attempt: attempt["value"])
123    lower = _candidate_lower_value(beta, radius_squared, best["action"])
124    upper = float(np.nextafter(lower+config["global_gap_tolerance"], np.inf))
125    certified = lcb_below_level(beta, radius_squared, upper)
126    return {"optimization_state": "certified" if certified else "uncertified",
127            "action": best["action"], "attempts": attempts,
128            "solver_success": best["success"], "candidate_value_lower": lower,
129            "global_value_upper": upper if certified else None,
130            "global_gap_upper": upper-lower if certified else None}

141def load_manifest(path):
142    """Validate the controlled sweep's fixed domains, axes, and seed streams."""
143    path = Path(path).resolve()
144    p = read_json(path)
145    if p.get("kind") != MANIFEST_KIND or p.get("domain") != "real_line":
146        raise ValueError("Require the controlled all-real bootstrap manifest.")
147    if not p["name"] or Path(p["name"]).name != p["name"] or p["name"] in {".", ".."}:
148        raise ValueError("Require a safe project name.")
149    if p["design"] != {"type": "iid_normal", "mean": 0.0, "std": 1.0}:
150        raise ValueError("This controlled experiment fixes standard normal training inputs.")
151    if p["truth"] != {"coefficients": [0., 5., -5.]}:
152        raise ValueError("This experiment fixes f(a)=5a-5a^2.")
153    if not 0 < p["delta"] < 1 or p["datasets"] < 2:
154        raise ValueError("Require 0<delta<1 and at least two datasets.")
155    for key, lower in (("N", 4), ("B", 2), ("sigma", 0)):
156        values = p["axes"][key]
157        if not values or values != sorted(set(values)) or p["baseline"][key] not in values:
158            raise ValueError("Axes must be sorted, unique, and contain the baseline.")
159        if any(not np.isfinite(v) or (v <= 0 if key == "sigma" else v < lower or int(v) != v) for v in values):
160            raise ValueError("Invalid sweep values.")
161    if set(p["seeds"]) != {"master", "design", "observation", "bootstrap"}:
162        raise ValueError("Separate named seed streams are required.")
163    if any(not isinstance(seed, int) or seed < 0 for seed in p["seeds"].values()):
164        raise ValueError("Seed roots must be nonnegative integers.")
165    if not 0 < p["supremum_tolerance"] < 1e-3:
166        raise ValueError("Invalid supremum tolerance.")
167    opt = p["optimizer"]
168    if opt["step_rule"] != "l-bfgs-b" or not opt["starts"] or any(not 0 <= a <= 1 for a in opt["starts"]):
169        raise ValueError("Require repository L-BFGS-B with starts in [0,1], without bounds.")
170    if any(not np.isfinite(opt[k]) or opt[k] <= 0 for k in ("t_steps", "gradient_tolerance", "ftol", "global_gap_tolerance")):
171        raise ValueError("Invalid optimizer tolerances.")
172    if p["launch"] != {"mode": "local", "array": "none"}:
173        raise ValueError("Use a local non-array launch.")
174    return SweepManifest(p["name"], p, path, PolicyLCBLaunchSpec(**p["launch"]))

184def dataset_streams(payload, index):
185    """Generate nested observations and bootstrap columns with independent seeds."""
186    seeds = {key: derive_seed(payload["seeds"][key], f"{payload['seeds']['master']}:dataset:{index}")
187             for key in ("design", "observation", "bootstrap")}
188    nmax, bmax = max(payload["axes"]["N"]), max(payload["axes"]["B"])
189    x = np.random.default_rng(seeds["design"]).normal(size=nmax)
190    epsilon = np.random.default_rng(seeds["observation"]).normal(size=nmax)
191    z = np.random.default_rng(seeds["bootstrap"]).normal(size=(bmax, nmax))
192    return seeds, x, epsilon, z

195def calibrate_design(x, z, tolerance):
196    """Calibrate standardized bootstrap refits using design and Gaussian draws only."""
197    p = features(x)
198    if len(x) <= 3 or np.linalg.matrix_rank(p) != 3:
199        raise ValueError("Require full-rank quadratic OLS with N>3.")
200    q, triangular = np.linalg.qr(p, mode="reduced")
201    inverse = solve_triangular(triangular, np.eye(3))
202    covariance = inverse @ inverse.T
203    d = solve_triangular(triangular, q.T @ z.T).T
204    cert = QuadraticErrorCertificate(covariance, 1.)
205    bounds = np.array([cert.supremum_interval(cert.difference(draw, np.zeros(3)), tolerance) for draw in d])
206    return p, q, triangular, covariance, d, bounds

216def run_dataset(manifest, index, runs_root, force=False):
217    """Compute a paired dataset block, saving sufficient statistics and provenance."""
218    started = time.monotonic()
219    payload = manifest.payload
220    project = Path(runs_root)/manifest.name
221    destination = project/"datasets"/f"dataset-{index:03d}"
222    contract = {"manifest": payload, "dataset": index, "sources": _source_contract()}
223    summary_path = destination/"summary.json"
224    if summary_path.exists() and not force:
225        previous = read_json(summary_path)
226        if previous["contract"] != contract or previous["arrays"] != file_record(destination/"draws.npz"):
227            raise ValueError("Cached sweep contract/artifacts changed; use a new name or explicit --force.")
228        return previous
229    seeds, x, epsilon, z = dataset_streams(payload, index)
230    beta0 = np.array(payload["truth"]["coefficients"])
231    reference = optimize_lcb(beta0, np.eye(3), 1., 0., payload["optimizer"])
232    if reference["optimization_state"] != "certified":
233        raise RuntimeError("Repository true-reference optimizer did not certify.")
234    a_star = reference["action"]
235    arrays = {"training_actions": x, "standardized_observation_errors": epsilon}
236    fits, solved, rows = {}, {}, []
237    for n in payload["axes"]["N"]:
238        bmax = max(payload["axes"]["B"]) if n == payload["baseline"]["N"] else payload["baseline"]["B"]
239        p, q, triangular, covariance, d, bounds = calibrate_design(x[:n], z[:bmax, :n], payload["supremum_tolerance"])
240        fits[n] = p, q, triangular, covariance, d, bounds
241        arrays.update({f"N{n}_covariance": covariance, f"N{n}_bootstrap_unit_perturbations": d,
242                       f"N{n}_bootstrap_supremum_bounds": bounds})
243    for axis, axis_value, setting in settings(payload):
244        n, sigma, b = setting["N"], setting["sigma"], setting["B"]
245        key = (n, sigma, b)
246        if key not in solved:
247            p, q, triangular, covariance, d, bounds = fits[n]
248            y = p @ beta0 + sigma*epsilon[:n]
249            beta = solve_triangular(triangular, q.T @ y)
250            residual = y-p @ beta
251            sigma_hat = float(np.linalg.norm(residual)/np.sqrt(n-3))
252            critical_low, critical = np.quantile(bounds[:b], 1-payload["delta"], axis=0, method="higher")
253            cert = QuadraticErrorCertificate(covariance, sigma_hat)
254            error = cert.difference(beta, beta0)
255            covered = cert.contains(error, float(critical))
256            observed_bounds = cert.supremum_interval(error, payload["supremum_tolerance"])
257            identity = solve_triangular(triangular, q.T @ (sigma*epsilon[:n]))
258            identity_error = float(np.max(np.abs(identity-(beta-beta0))))
259            if identity_error > 1e-9*max(1., np.max(np.abs(identity))):
260                raise RuntimeError("OLS error identity failed.")
261            outcome = optimize_lcb(beta, covariance, sigma_hat, float(critical), payload["optimizer"])
262            p_star = features(np.array([a_star]))[0]
263            width = float(critical*sigma_hat*np.sqrt(p_star @ covariance @ p_star))
264            valid = outcome["optimization_state"] == "certified"
265            regret = float(p_star @ beta0-features(np.array([outcome["action"]]))[0] @ beta0) if valid else None
266            regret_bound = 2*width
267            if valid and covered and regret > regret_bound+outcome["global_gap_upper"]+reference["global_gap_upper"]:
268                raise RuntimeError("Covered certified optimizer violated the regret inequality.")
269            prefix = f"N{n}_sd{sigma:g}_B{b}"
270            h = cert.threshold_polynomial(error**2, float(critical))
271            arrays.update({f"{prefix}_y": y, f"{prefix}_beta_hat": beta,
272                           f"{prefix}_error_coefficients": beta-beta0,
273                           f"{prefix}_bootstrap_beta": beta+sigma_hat*d[:b],
274                           f"{prefix}_containment_exact_coefficients": np.array([str(h.nth(i)) for i in range(5)])})
275            solved[key] = {
276                "dataset": index, "N": n, "sigma": sigma, "B": b,
277                "covered": covered, "width_at_true_optimum": width,
278                "regret": regret, "regret_bound": regret_bound,
279                "critical_value": float(critical), "critical_value_lower": float(critical_low),
280                "sigma_hat": sigma_hat, "sigma2_hat": sigma_hat**2,
281                "observed_statistic_lower": observed_bounds[0], "observed_statistic_upper": observed_bounds[1],
282                "identity_error": identity_error, "true_action": a_star,
283                "lcb_action": outcome["action"], "optimization_state": outcome["optimization_state"],
284                "solver_success": outcome.get("solver_success"), "global_gap_upper": outcome["global_gap_upper"],
285                "optimizer_attempts": outcome["attempts"], "array_prefix": prefix,
286            }
287        rows.append({**solved[key], "axis": axis, "axis_value": axis_value})
288    destination.mkdir(parents=True, exist_ok=True)
289    np.savez_compressed(destination/"draws.npz", **arrays)
290    summary = {"contract": contract, "seeds": seeds, "reference": reference, "rows": rows,
291               "arrays": file_record(destination/"draws.npz"), "elapsed_seconds": time.monotonic()-started}
292    write_json_atomic(summary_path, summary)
293    print(f"Dataset {index+1}/{payload['datasets']}: {summary['elapsed_seconds']:.1f}s", flush=True)
294    return summary

302def aggregate(rows):
303    """Aggregate independent datasets; never hide missing regret denominators."""
304    result = []
305    for axis, value in dict.fromkeys((r["axis"], r["axis_value"]) for r in rows):
306        group = [r for r in rows if r["axis"] == axis and r["axis_value"] == value]
307        count, covered = len(group), sum(r["covered"] for r in group)
308        lower, upper = wilson_interval(covered, count)
309        row = {"axis": axis, "axis_value": value, "datasets": count,
310               "N": group[0]["N"], "sigma": group[0]["sigma"], "B": group[0]["B"],
311               "covered_count": covered, "coverage": covered/count,
312               "coverage_lower": lower, "coverage_upper": upper}
313        for metric in ("width_at_true_optimum", "regret", "regret_bound", "critical_value"):
314            row[f"{metric}_mean"], row[f"{metric}_se"] = _mean_se([r[metric] for r in group if r[metric] is not None])
315        for state in ("certified", "unbounded", "degenerate_tail", "optimizer_failed", "uncertified"):
316            row[f"{state}_count"] = sum(r["optimization_state"] == state for r in group)
317        row["solver_warning_count"] = sum(r["solver_success"] is False for r in group)
318        result.append(row)
319    return result

def optimize_lcb(beta, covariance, sigma_hat, critical, config): 
 93def optimize_lcb(beta, covariance, sigma_hat, critical, config):
 94    """Use repository multistart optimization and certify its all-real gap."""
 95    cert = QuadraticErrorCertificate(covariance, sigma_hat)
 96    radius_squared = cert.q.mul_ground(sp.Rational(float(critical))**2)
 97    leading = sp.Rational(float(beta[2]))
 98    tail_square = radius_squared.nth(4)
 99    # Comparing squares is exact when beta_2 >= 0; negative beta_2 is coercive.
100    if leading >= 0 and leading**2 >= tail_square:
101        return {"optimization_state": "unbounded" if leading**2 > tail_square else "degenerate_tail",
102                "attempts": [], "action": None, "global_gap_upper": None}
103    objective = QuadraticLCBObjective(beta, covariance, critical*sigma_hat)
104    attempts = []
105    for start in config["starts"]:
106        theta, trace = run_first_order_minimize(
107            np.array([start], dtype=float), np.zeros((1, 1)), objective,
108            t_steps=config["t_steps"], n_grad_samples=1, sigma=0.,
109            algorithm=config["step_rule"], grad_norm_tol=config["gradient_tolerance"],
110            ftol=config["ftol"],
111        )
112        a = float(theta[0])
113        attempts.append({"start": start, "action": a,
114                         "value": float(-objective.value(theta, None)),
115                         "success": bool(trace.optimizer_success),
116                         "status": trace.optimizer_status, "message": trace.optimizer_message,
117                         "gradient_norm": float(np.linalg.norm(objective.grad(theta, None)))})
118    finite = [attempt for attempt in attempts if np.isfinite([attempt["action"], attempt["value"]]).all()]
119    if not finite:
120        return {"optimization_state": "optimizer_failed", "attempts": attempts,
121                "action": None, "global_gap_upper": None}
122    # Select only among actual repository solver outputs, never grid/root actions.
123    best = max(finite, key=lambda attempt: attempt["value"])
124    lower = _candidate_lower_value(beta, radius_squared, best["action"])

130            "global_value_upper": upper if certified else None,
