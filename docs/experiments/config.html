 41def _x_fixed_frame_matches_state_dim(objective: object, x_fixed_frame: Any, state_dim: int) -> bool:
 42    state_cols = tuple(getattr(objective, "acceptance_state_cols", ()))
 43    if not state_cols:
 44        return x_fixed_frame.shape[1] == state_dim
 45    acceptance_model = getattr(objective, "acceptance_model", None)
 46    auxiliary_cols = tuple(getattr(acceptance_model, "auxiliary_state_cols", ()))
 47    policy_feature_cols = tuple(getattr(objective, "policy_feature_cols", ()) or ())
 48    expected_cols = set(state_cols) | set(auxiliary_cols) | set(policy_feature_cols)
 49    if getattr(objective, "loss_source", "predicted") == "observed":
 50        observed_loss_col = getattr(objective, "observed_loss_col", None)
 51        if observed_loss_col is None:
 52            return False
 53        expected_cols.add(str(observed_loss_col))
 54    return (
 55        len(state_cols) == state_dim
 56        and set(x_fixed_frame.columns) == expected_cols
 57        and x_fixed_frame.shape[1] == len(expected_cols)
 58    )

597            "loss_source": objective.loss_source,
598            "observed_loss_col": objective.observed_loss_col,

600            "u_bounds": list(objective.u_bounds) if objective.u_bounds is not None else None,
601            "acceptance_floor": float(objective.acceptance_floor)
602            if objective.acceptance_floor is not None
603            else None,
604            "acceptance_penalty_weight": float(objective.acceptance_penalty_weight)
605            if objective.acceptance_penalty_weight is not None
606            else None,
607            "acceptance_penalty_temperature": float(objective.acceptance_penalty_temperature),
608            "lagrangian_lambda": float(objective.lagrangian_lambda)
