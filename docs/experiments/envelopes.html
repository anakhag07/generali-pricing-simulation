 20    def __init__(self, X, actions, *, action_degree=1, supremum_tolerance=1e-8):
 21        X, actions = locations(X, actions)
 22        if (isinstance(action_degree, bool) or action_degree not in (1, 2)
 23                or (action_degree == 2 and X.shape[1] != 0)):
 24            raise ValueError("Require contextual affine or action-only quadratic features.")
 25        if not np.isfinite(supremum_tolerance) or not 0 < supremum_tolerance < 1e-3:
 26            raise ValueError("Require a positive small supremum tolerance.")
 27        self.action_degree = action_degree
 28        self.supremum_tolerance = float(supremum_tolerance)
 29        self.X = np.array(X, copy=True)
 30        self.design = self.features(X, actions)
 31        n, p = self.design.shape
 32        if n <= p or np.linalg.matrix_rank(self.design) != p:
 33            raise ValueError("Wild OLS requires full column rank and n > feature dimension.")
 34        self._orthogonal, self._triangular = np.linalg.qr(self.design, mode="reduced")
 35        inverse = solve_triangular(self._triangular, np.eye(p))
 36        self.covariance = inverse @ inverse.T
 37        self.leverage = np.sum(self._orthogonal**2, axis=1)
 38        for array in (self.X, self.design, self.covariance, self.leverage):
 39            array.flags.writeable = False
 40        if action_degree == 2:
 41            self._certificate = QuadraticErrorCertificate(self.covariance, 1.)
 42            return
 43        # With no contexts every action fiber is identical; retain all rows only
 44        # for fitting and observed leverage, not for repeated supremum checks.
 45        self._contexts = self.design[:1, :-1] if X.shape[1] == 0 else self.design[:, :-1]
 46        self._cross = self._contexts @ self.covariance[:-1, -1]
 47        self._constant = np.einsum(
 48            "ij,jk,ik->i", self._contexts, self.covariance[:-1, :-1], self._contexts,
 49        )
 50        self._action_variance = self.covariance[-1, -1]
 51        self._schur = self._constant - self._cross**2 / self._action_variance
 52        if (not np.isfinite(self.covariance).all() or self._action_variance <= 0
 53                or np.any(self._schur <= 0)):
 54            raise ValueError("Numerically singular all-real design geometry.")

 56    def features(self, X, actions):
 57        """Evaluate the fitted basis exactly at paired inputs, including off-grid actions."""
 58        X, actions = locations(X, actions)
 59        if X.shape[1] != self.X.shape[1]:
 60            raise ValueError("Design and evaluation context dimensions disagree.")
 61        phi = features(X, actions)
 62        return np.column_stack((phi, actions**2)) if self.action_degree == 2 else phi

 65        """Fit the auxiliary OLS coefficients using the cached QR factors."""
 66        y = values(y, len(self.design), "responses")
 67        return solve_triangular(self._triangular, self._orthogonal.T @ y)

 69    def bootstrap_errors(self, y, multipliers, *, predictor=None):
 70        """Fit and bootstrap Y-predictor; omitted predictor retains raw-response replay."""
 71        y = values(y, len(self.design), "responses")
 72        if predictor is not None:
 73            y = y - values(predictor, len(self.design), "predictor")
 74        multipliers = np.asarray(multipliers)
 75        if (multipliers.ndim != 2 or multipliers.shape[1] != len(self.design)
 76                or len(multipliers) == 0 or np.iscomplexobj(multipliers)
 77                or not np.all((multipliers == -1) | (multipliers == 1))):
 78            raise ValueError("Multipliers must have shape (B,n) with values -1 or +1.")
 79        beta = self.fit(y)
 80        residuals = y - self.design @ beta
 81        # This is exactly refitting y*=P beta+residuals*xi on unchanged P.
 82        errors = solve_triangular(
 83            self._triangular, self._orthogonal.T @ (multipliers * residuals).T,
 84        ).T
 85        return beta, errors

 87    def statistics(self, errors):
 88        """Return all-real statistics, using certified upper bounds for quadratic features."""
 89        errors = np.asarray(errors, dtype=float)
 90        if (errors.ndim != 2 or errors.shape[1] != self.design.shape[1]
 91                or not np.isfinite(errors).all()):
 92            raise ValueError("Coefficient errors must have finite shape (B,p).")
 93        if self.action_degree == 2:
 94            return self.statistic_bounds(errors)[:, 1]
 95        intercepts = errors[:, :-1] @ self._contexts.T
 96        intercepts -= errors[:, -1, None] * (self._cross / self._action_variance)
 97        # Schur-complement form is a sum of nonnegative squares and includes
 98        # the projective limit in both action tails, without selecting an action.
 99        squared = intercepts**2 / self._schur
100        squared += errors[:, -1, None]**2 / self._action_variance
101        return np.sqrt(np.max(squared, axis=1))

103    def statistic_bounds(self, errors):
104        """Bracket quadratic suprema; affine entries repeat the coefficient statistic."""
105        errors = np.asarray(errors, dtype=float)
106        if (errors.ndim != 2 or errors.shape[1] != self.design.shape[1]
107                or not np.isfinite(errors).all()):
108            raise ValueError("Coefficient errors must have finite shape (B,p).")
109        if self.action_degree == 1:
110            statistic = self.statistics(errors)
111            return np.column_stack((statistic, statistic))
112        return np.array([
113            self._certificate.supremum_interval(
114                self._certificate.difference(error, np.zeros(3)), self.supremum_tolerance,
115            ) for error in errors
116        ]).reshape(-1, 2)

118    def residual_components(self, X, actions, beta, critical):
119        r"""Evaluate $$r_{lower}=[w-\widetilde g]_+,\ r_{upper}=[w+\widetilde g]_+.$$"""
120        phi = self.features(X, actions)
121        beta = values(beta, self.design.shape[1], "residual coefficients")
122        critical = _critical(critical)
123        residual = phi @ beta
124        # QR coordinates avoid cancellation in the geometric width.
125        width = critical * np.linalg.norm(
126            solve_triangular(self._triangular.T, phi.T, lower=True), axis=0,
127        )
128        lower, upper = residual_radii(residual, width)
129        return {"residual_fit": residual, "width": width,
130                "radius_lower": lower, "radius_upper": upper}

133        """Check the residual-fit band jointly over the entire target action domain."""
134        p = self.design.shape[1]
135        beta = values(beta, p, "residual coefficients")

143    def residual_contains(self, beta, residual_truth, critical):
144        """Check asymmetric transferred containment using signed residual inequalities."""
145        p = self.design.shape[1]
146        beta = values(beta, p, "residual coefficients")
147        residual_truth = values(residual_truth, p, "residual truth")
148        critical = _critical(critical)
149        if np.all(residual_truth == 0):
150            return True  # The transferred band always includes its fixed predictor.
151        if self.action_degree == 2:
152            g = self._certificate.difference(residual_truth, np.zeros(3))
153            fitted = self._certificate.difference(beta, np.zeros(3))
154            return _residual_polynomials_contain(g, fitted, self._certificate.q, critical)
155        for context in self._contexts:
156            phi0 = np.r_[context, 0.]
157            phi1 = np.eye(p)[-1]
158            q = np.array([phi0 @ self.covariance @ phi0,
159                          2*phi0 @ self.covariance @ phi1,
160                          self.covariance[-1, -1]], dtype=np.longdouble)
161            g = np.array([phi0 @ residual_truth, residual_truth[-1]], dtype=np.longdouble)
162            fitted = np.array([phi0 @ beta, beta[-1]], dtype=np.longdouble)
163            result = _residual_fiber_contains(g, fitted, critical**2*q)
164            if result is None:
165                basis = sp.Matrix([sp.Rational(float(x)) for x in phi0])
166                basis[-1] = VARIABLE
167                covariance = sp.Matrix(self.covariance).applyfunc(lambda x: sp.Rational(float(x)))
168                poly = lambda v: sp.Poly(sum(b*sp.Rational(float(c)) for b, c in zip(basis, v)),
169                                        VARIABLE, domain=sp.QQ)
170                q_exact = sp.Poly((basis.T*covariance*basis)[0], VARIABLE, domain=sp.QQ)
171                result = _residual_polynomials_contain(poly(residual_truth), poly(beta), q_exact, critical)
172            if not result:
173                return False
174        return True

176    def envelope(self, predictor, beta, critical, metadata=None):
177        """Replay the historical affine function-target symmetric transfer."""
178        if self.action_degree != 1:
179            raise ValueError("Historical symmetric transfer requires affine features.")
180        beta = values(beta, self.design.shape[1], "coefficients").copy()
181        critical = _critical(critical)
182        auxiliary_model = LinearModel(beta)
183        covariance = self.covariance
184
185        def components(X, actions):
186            phi = features(X, actions)
187            if phi.shape[1] != len(beta):
188                raise ValueError("Envelope and evaluation context dimensions disagree.")
189            auxiliary = auxiliary_model(X, actions)
190            leverage = np.einsum("ij,jk,ik->i", phi, covariance, phi)
191            if not np.isfinite(leverage).all() or np.any(leverage < 0):
192                raise ValueError("Numerical leverage evaluation is negative or nonfinite.")
193            return {"auxiliary": auxiliary, "width": np.sqrt(critical**2 * leverage),
194                    "discrepancy": np.abs(values(predictor(X, actions), len(actions), "predictor") - auxiliary)}
195
196        def radius(X, actions):
197            parts = components(X, actions)
198            return parts["discrepancy"] + parts["width"]
199
200        return FittedEnvelope(
201            center=predictor, radius=radius, components=components,
202            state={"beta": beta, "covariance": covariance, "scale_squared": critical**2},
203            metadata={"construction": "linear_wild_bootstrap_transfer", "critical": critical,
204                      "residuals": "raw", "multipliers": "Rademacher",
205                      **({} if metadata is None else metadata)},
206        )

208    def transferred_contains(self, beta, predictor_coeff, truth_coeff, critical):
209        """Check the historical affine symmetric transfer, not the residual construction."""
210        if self.action_degree != 1:
211            raise ValueError("Historical symmetric transfer requires affine features.")
212        p = self.design.shape[1]
213        beta = values(beta, p, "coefficients")
214        predictor_coeff = values(predictor_coeff, p, "predictor coefficients")
215        truth_coeff = values(truth_coeff, p, "truth coefficients")
216        critical = _critical(critical)
217        if np.array_equal(predictor_coeff, truth_coeff):
218            return True
219        # Extended precision reduces cancellation before the boundary guard;
220        # exact fallback treats the stored floats as rationals, as linear.py does.
221        dtype = np.longdouble
222        contexts = self._contexts.astype(dtype)
223        h = predictor_coeff.astype(dtype) - truth_coeff.astype(dtype)
224        g = predictor_coeff.astype(dtype) - beta.astype(dtype)
225        h0, g0 = contexts @ h[:-1], contexts @ g[:-1]
226        covariance = self.covariance.astype(dtype)
227        q0 = np.einsum("ij,jk,ik->i", contexts, covariance[:-1, :-1], contexts)
228        q1 = 2 * (contexts @ covariance[:-1, -1])
229        q2 = covariance[-1, -1]
230        scale = dtype(float(critical**2))
231        for index in range(len(contexts)):
232            result = _fiber_contains((h0[index], h[-1]), (g0[index], g[-1]),
233                                     scale * np.array([q0[index], q1[index], q2], dtype=dtype))
234            if result is None:
235                envelope = self.envelope(LinearModel(predictor_coeff), beta, critical)
236                exact = check_linear_containment(
237                    envelope, LinearModel(truth_coeff), TargetDomain(self.X[index:index+1]),
238                )
239                result = exact.status == "covered"
240            if not result:
241                return False
242        return True

def evaluate_tightness(envelope, truth, X, actions, weights=None): 
168def evaluate_tightness(envelope, truth, X, actions, weights=None):
169    """Return pointwise radius/excess and summaries over a declared finite measure."""
170    X, actions = locations(X, actions)
171    evaluated = envelope.evaluate(X, actions)
172    actual = values(truth(X, actions), len(X), "truth")
173    weight = np.ones(len(X)) if weights is None else values(weights, len(X), "weights")
174    if np.any(weight < 0) or weight.sum() <= 0:
175        raise ValueError("Diagnostic weights must be nonnegative with positive sum.")
176    weight = weight / weight.sum()
177    error = np.abs(evaluated.center - actual)
178    points = {"actions": actions, "center": evaluated.center, "truth": actual,
179              "radius": evaluated.radius, "absolute_error": error,
180              "excess": evaluated.radius-error, "weights": weight}

182    summary = {}
183    for key in ("radius", "excess", "absolute_error"):
184        value = points[key]
185        # Inverse weighted empirical CDF; this is a diagnostic quantile, not an action selection.
186        order = np.argsort(value)
187        positive = weight[order] > 0
188        ordered, cumulative = value[order][positive], np.cumsum(weight[order][positive])
189        summary[f"{key}_mean"] = float(weight @ value)
190        for q in (.5, .9):
191            index = min(int(np.searchsorted(cumulative, q, side="left")), len(ordered)-1)
192            summary[f"{key}_q{int(100*q)}"] = float(ordered[index])
193    summary["negative_excess_weight"] = float(weight @ (points["excess"] < 0))
194    return points, summary
