 13def wilson_interval(
 14    successes: int,
 15    trials: int,
 16    *,
 17    confidence: float = 0.95,
 18) -> tuple[float, float]:
 19    """Return a Wilson score interval with exact endpoints at all-fail/all-pass."""
 20    if trials <= 0:
 21        raise ValueError("trials must be positive.")
 22    z_value = float(norm.ppf(0.5 + confidence / 2.0))
 23    proportion = successes / trials
 24    denominator = 1.0 + z_value**2 / trials
 25    center = (proportion + z_value**2 / (2.0 * trials)) / denominator
 26    radius = (
 27        z_value
 28        * np.sqrt(proportion * (1.0 - proportion) / trials + z_value**2 / (4.0 * trials**2))
 29        / denominator
 30    )
 31    low = 0.0 if successes == 0 else max(0.0, center - radius)
 32    high = 1.0 if successes == trials else min(1.0, center + radius)
 33    return float(low), float(high)

 39def locations(X, actions):
 40    """Validate paired finite contexts/actions, allowing zero context columns."""
 41    X, actions = np.asarray(X, dtype=float), np.asarray(actions, dtype=float)
 42    if (X.ndim != 2 or len(X) == 0
 43            or actions.shape != (len(X),)
 44            or not np.isfinite(X).all() or not np.isfinite(actions).all()):
 45        raise ValueError("Require finite X of shape (n,d) and actions of shape (n,).")
 46    return X, actions

 49def values(result, size, name):
 50    """Require one finite model value per queried context/action pair."""
 51    result = np.asarray(result, dtype=float)
 52    if result.shape != (size,) or not np.isfinite(result).all():
 53        raise ValueError(f"{name} must return {size} finite values, shape ({size},).")
 54    return result

 75class TargetDomain:
 76    """Observed context fibers crossed with R or an explicitly finite action set."""
 77
 78    contexts: np.ndarray
 79    actions: np.ndarray | None = None
 80
 81    def __post_init__(self):
 82        X = np.asarray(self.contexts, dtype=float)
 83        locations(X, np.zeros(len(X)))
 84        X = X.copy()
 85        X.flags.writeable = False
 86        object.__setattr__(self, "contexts", X)

 88            a = np.asarray(self.actions, dtype=float)

 91            a = a.copy()
 92            a.flags.writeable = False

 97        """Return the scope of the simultaneous containment event."""
 98        return "real_line" if self.actions is None else "finite_set"

100    def points(self):
101        """Expand a finite target into paired evaluation locations."""
102        if self.actions is None:
103            raise ValueError("The real line cannot be expanded into a finite target.")
104        return cartesian_locations(self.contexts, self.actions)

107def cartesian_locations(contexts, actions):
108    """Expand contexts by actions in context-major order."""
109    X, a = np.asarray(contexts, dtype=float), np.asarray(actions, dtype=float)
110    return locations(np.repeat(X, len(a), axis=0), np.tile(a, len(X)))

135class FittedEnvelope:
136    """Deterministic fitted callables with replayable numeric state and metadata."""
137
138    center: Model
139    radius: Model | None = None
140    state: dict[str, Any] = field(default_factory=dict)
141    metadata: dict[str, Any] = field(default_factory=dict)
142    components: Callable | None = None
143    radii: Callable | None = None

145
146    def evaluate(self, X, actions):
147        """Evaluate paired queries and reject invalid or negative radii."""
148        X, actions = locations(X, actions)
149        center = values(self.center(X, actions), len(X), "center")
150        if (self.radius is None) == (self.radii is None):
151            raise ValueError("Specify either a symmetric radius or separate lower/upper radii.")
152        if self.radii is None:
153            radius = values(self.radius(X, actions), len(X), "radius")
154            lower = upper = radius
155        else:
156            radius = None
157            lower, upper = self.radii(X, actions)
158            lower = values(lower, len(X), "lower radius")
159            upper = values(upper, len(X), "upper radius")
160        if np.any(lower < 0) or np.any(upper < 0):
161            raise ValueError("Envelope radii must be nonnegative.")
162        parts = {} if self.components is None else self.components(X, actions)
163        if not isinstance(parts, Mapping):
164            raise ValueError("Envelope components must be a mapping.")
165        parts = {key: values(value, len(X), f"component {key}") for key, value in parts.items()}
166        return EnvelopeValues(center, radius, parts, lower, upper)
167
168    def derivative(self, X, actions, endpoint):
169        """Return a declared endpoint action derivative, rejecting query-only methods."""
170        if endpoint not in {"center", "lower", "upper"}:
171            raise ValueError("Endpoint must be center, lower or upper.")
172        if self.action_derivatives is None:
173            raise ValueError("This fitted band does not provide first-order derivatives.")
174        X, actions = locations(X, actions)
175        return values(self.action_derivatives(X, actions)[endpoint], len(X), "action derivative")

222def evaluate_tightness(envelope, truth, X, actions, weights=None):
223    """Return pointwise radius/excess and summaries over a declared finite measure."""
224    X, actions = locations(X, actions)
225    evaluated = envelope.evaluate(X, actions)
226    actual = values(truth(X, actions), len(X), "truth")
227    weight = np.ones(len(X)) if weights is None else values(weights, len(X), "weights")
228    if np.any(weight < 0) or weight.sum() <= 0:
229        raise ValueError("Diagnostic weights must be nonnegative with positive sum.")
230    weight = weight / weight.sum()
231    error = np.abs(evaluated.center - actual)
232    half_width = evaluated.radius if evaluated.radius is not None else (evaluated.upper-evaluated.lower)/2
233    excess = (evaluated.radius-error if evaluated.radius is not None else
234              np.minimum(actual-evaluated.lower, evaluated.upper-actual))
235    points = {"actions": actions, "center": evaluated.center, "truth": actual,
236              "radius": half_width, "absolute_error": error,
237              "excess": excess, "weights": weight}
238    if evaluated.radius is None:
239        points.update(lower=evaluated.lower, upper=evaluated.upper,
240                      radius_lower=evaluated.radius_lower, radius_upper=evaluated.radius_upper)
241    points.update({f"component_{key}": value for key, value in evaluated.components.items()})
242    summary = {}
243    for key in ("radius", "excess", "absolute_error"):
244        value = points[key]
245        # Inverse weighted empirical CDF; this is a diagnostic quantile, not an action selection.
246        order = np.argsort(value)
247        positive = weight[order] > 0
248        ordered, cumulative = value[order][positive], np.cumsum(weight[order][positive])
249        summary[f"{key}_mean"] = float(weight @ value)
250        for q in (.5, .9):
251            index = min(int(np.searchsorted(cumulative, q, side="left")), len(ordered)-1)
252            summary[f"{key}_q{int(100*q)}"] = float(ordered[index])
253    summary["negative_excess_weight"] = float(weight @ (points["excess"] < 0))
254    return points, summary

257def check_finite_containment(envelope, truth, target):
258    """Check precisely the declared finite set, without claiming continuum coverage."""
259    X, a = target.points()
260    evaluated = envelope.evaluate(X, a)
261    actual = values(truth(X, a), len(a), "truth")
262    margin = (evaluated.radius-np.abs(evaluated.center-actual) if evaluated.radius is not None else
263              np.minimum(actual-evaluated.lower, evaluated.upper-actual))
264    failed = np.flatnonzero(margin < 0)
265    covered = (margin >= 0).reshape(len(target.contexts), -1).all(axis=1)
266    details = {
267        "point_count": len(a), "context_count": len(target.contexts),
268        "contexts": [{"context_index": index, "covered": bool(value)}
269                     for index, value in enumerate(covered)],
270        "numerics": "floating-point evaluation; no rounding enclosure",
271    }
272    if len(failed):
273        index = int(failed[0])
274        details["witness"] = {"context": X[index].tolist(), "action": float(a[index]),
275                              "excess": float(margin[index])}
276    return ContainmentResult("violated" if len(failed) else "covered",
277                             "finite_evaluation", target.kind, details)

280def aggregate_coverage(records: Sequence[dict], expected: int, confidence=.95):
281    """Retain every repetition; suppress estimates with missing or unresolved cases."""
282    if expected < 1 or len(records) > expected or not 0 < confidence < 1:
283        raise ValueError("Invalid coverage aggregation dimensions or confidence.")
284    counts = {key: 0 for key in ("covered", "violated", "unknown", "fit_failed")}
285    for row in records:
286        status = row["status"]
287        if status not in counts:
288            raise ValueError(f"Unrecognized coverage status: {status}")
289        counts[status] += 1
290    missing = expected-len(records)
291    decisive = counts["covered"]+counts["violated"] == expected
292    interval = wilson_interval(counts["covered"], expected, confidence=confidence) if decisive else None
293    return {"kind": "empirical_estimate", "requested": expected, "counts": counts,
294            "missing": missing, "estimate": counts["covered"]/expected if decisive else None,
295            "interval": interval, "interval_method": "Wilson" if decisive else None,
296            "confidence": confidence, "conditioning": "fixed logged contexts/actions and frozen predictor"}

def evaluate_tightness(envelope, truth, X, actions, weights=None): 
223def evaluate_tightness(envelope, truth, X, actions, weights=None):
224    """Return pointwise radius/excess and summaries over a declared finite measure."""
225    X, actions = locations(X, actions)
226    evaluated = envelope.evaluate(X, actions)
227    actual = values(truth(X, actions), len(X), "truth")
228    weight = np.ones(len(X)) if weights is None else values(weights, len(X), "weights")
229    if np.any(weight < 0) or weight.sum() <= 0:
230        raise ValueError("Diagnostic weights must be nonnegative with positive sum.")
231    weight = weight / weight.sum()
232    error = np.abs(evaluated.center - actual)
233    half_width = evaluated.radius if evaluated.radius is not None else (evaluated.upper-evaluated.lower)/2
234    excess = (evaluated.radius-error if evaluated.radius is not None else
235              np.minimum(actual-evaluated.lower, evaluated.upper-actual))
236    points = {"actions": actions, "center": evaluated.center, "truth": actual,
237              "radius": half_width, "absolute_error": error,
238              "excess": excess, "weights": weight}
239    if evaluated.radius is None:
240        points.update(lower=evaluated.lower, upper=evaluated.upper,
241                      radius_lower=evaluated.radius_lower, radius_upper=evaluated.radius_upper)
242    points.update({f"component_{key}": value for key, value in evaluated.components.items()})
243    summary = {}
244    for key in ("radius", "excess", "absolute_error"):
245        value = points[key]
246        # Inverse weighted empirical CDF; this is a diagnostic quantile, not an action selection.
247        order = np.argsort(value)
248        positive = weight[order] > 0
249        ordered, cumulative = value[order][positive], np.cumsum(weight[order][positive])
250        summary[f"{key}_mean"] = float(weight @ value)
251        for q in (.5, .9):
252            index = min(int(np.searchsorted(cumulative, q, side="left")), len(ordered)-1)
253            summary[f"{key}_q{int(100*q)}"] = float(ordered[index])
254    summary["negative_excess_weight"] = float(weight @ (points["excess"] < 0))
255    return points, summary
