121def _coverage_statistics(values):
122    valid = [value for value in values if value is not None]
123    count = sum(valid)
124    lo, hi = wilson_interval(count, len(valid)) if valid else (None, None)
125    return {'count': count if valid else None, 'coverage': count/len(valid) if valid else None,
126            'wilson_lower': lo, 'wilson_upper': hi, 'valid_repetitions': len(valid),
127            'unavailable_repetitions': len(values)-len(valid),
128            'full_run_lower': count/len(values),
129            'full_run_upper': (count+len(values)-len(valid))/len(values)}

396def _objective_regret(truth, contexts, true_actions, actions):
397    optimal = float(np.mean(truth(contexts, true_actions)))
398    converged = float(np.mean(truth(contexts, actions)))
399    regret = optimal-converged
400    rms = float(np.sqrt(np.mean((actions-.5)**2)))
401    if not np.isclose(regret, .5*rms*rms, atol=1e-12, rtol=1e-10):
402        raise ValueError('Direct objective regret disagrees with the quadratic identity.')
403    return {'objective_at_theta_star': optimal, 'objective_at_theta_hat': converged,
404            'regret': regret, 'rms_action_error': rms}
