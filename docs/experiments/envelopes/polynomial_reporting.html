 16def collect_polynomial(project):
 17    """Verify saved artifacts and report them without fitting, sampling or optimization."""
 18    project = Path(project)
 19    contract = read_json(project/'contract.json')
 20    digest = contract['sha256']
 21    p = contract['contract']['manifest']
 22    _verify(contract['environment'])
 23    for source in contract['source_snapshot']:
 24        _verify(source)
 25    # Queries must still use the original verified mathematical implementation.
 26    math_source = next(s for s in contract['contract']['sources'] if s['path'].endswith('/envelopes/polynomial.py'))
 27    if math_source['sha256'] != file_record(Path(__file__).with_name('polynomial.py'))['sha256']:
 28        raise ValueError('Numerical query implementation differs from the fitted-state contract.')
 29    with np.load(project/'inputs.npz', allow_pickle=False) as saved:
 30        snapshot = dict(saved)
 31    repetitions = []
 32    for index in range(p['repetitions']):
 33        path = project/'repetitions'/f'rep-{index:03d}.json'
 34        record = read_json(path)
 35        if record['contract_sha256'] != digest or record['index'] != index:
 36            raise ValueError('Incompatible repetition contract/index.')
 37        _verify(record['arrays'])
 38        repetitions.append(record)
 39    with np.load(project/'repetitions/rep-000.npz', allow_pickle=False) as saved:
 40        state = dict(saved)
 41    reference = read_json(project/'policies/truth.json')
 42    if reference['contract_sha256'] != digest:
 43        raise ValueError('Incompatible truth reference.')
 44    _verify(reference['arrays'])
 45    with np.load(reference['arrays']['path'], allow_pickle=False) as saved:
 46        true_actions = saved['actions']
 47    truth, residual = _models(p)
 48    mean, variance = p['design']['action_mean'], p['design']['action_variance']
 49    span = p['reporting']['action_window_standard_deviations']*np.sqrt(variance)
 50    grid = np.linspace(mean-span, mean+span, p['reporting']['action_points'])
 51    coordinates = {'action': grid, 'mean_true_residual': residual(snapshot['X'], grid[:, None]).mean(axis=1)}
 52    rows, policy_records = [], []
 53    for degree in p['feature_degrees']:
 54        band = restore_band(snapshot, state, degree)
 55        metrics = residual_metrics(band, residual, mean, variance)
 56        widths = integrated_width(band, mean, variance, p['reporting']['integration_tolerance'])
 57        policy = read_json(project/'policies'/f'degree-{degree}.json')
 58        if policy['contract_sha256'] != digest:
 59            raise ValueError('Incompatible policy contract.')
 60        _verify(policy['arrays'])
 61        for attempt in policy['attempts']:
 62            _verify(attempt['arrays'])
 63        with np.load(policy['arrays']['path'], allow_pickle=False) as saved:
 64            actions = saved['actions']
 65        objective_metrics = _objective_regret(truth, snapshot['X'], true_actions, actions)
 66        regret = objective_metrics['regret']
 67        row = {'degree': degree, 'feature_dimension': 4+degree, 'headline_repetition': 0,
 68               **metrics, **widths, **objective_metrics,
 69               'mean_action': float(actions.mean()), 'min_action': float(actions.min()), 'max_action': float(actions.max()),
 70               'max_leverage': float(state[f'degree{degree}_leverage'].max()),
 71               'design_condition_number': float(state[f'degree{degree}_condition_number']),
 72               'critical': float(state[f'degree{degree}_critical']),
 73               'solver_success_contexts': policy['selected_success_count'],
 74               'global_gap_certified_contexts': policy['global_gap_certified_count'],
 75               'global_gap_tolerance': policy['global_gap_tolerance'], 'contexts': p['n'],
 76               'coverage_repetitions': p['repetitions']}
 77        for event in ('coefficient', 'auxiliary', 'transferred'):
 78            count = sum(int(rep['events'][str(degree)][event]) for rep in repetitions)
 79            lo, hi = wilson_interval(count, p['repetitions'])
 80            row.update({f'{event}_count': count, f'{event}_coverage': count/p['repetitions'],
 81                        f'{event}_wilson_lower': lo, f'{event}_wilson_upper': hi})
 82        rows.append(row)
 83        policy_records.append(policy)
 84        fitted, width, lower, upper = band.evaluate(grid[:, None])
 85        coordinates.update({f'degree{degree}_mean_fit': fitted.mean(axis=1),
 86                            f'degree{degree}_mean_half_width': width.mean(axis=1),
 87                            f'degree{degree}_mean_lower_radius': lower.mean(axis=1),
 88                            f'degree{degree}_mean_upper_radius': upper.mean(axis=1),
 89                            f'degree{degree}_mean_full_width': (lower+upper).mean(axis=1),
 90                            f'degree{degree}_actions': actions})
 91        print(f"Degree {degree}: R2={metrics['r_squared']:.5f}, RMSE={metrics['rmse']:.5f}, "
 92              f"coverage={row['transferred_count']}/{p['repetitions']}, regret={regret:.6f}", flush=True)
 93    with (project/'summary.csv').open('w', newline='') as stream:
 94        writer = csv.DictWriter(stream, list(rows[0]))
 95        writer.writeheader()
 96        writer.writerows(rows)
 97    save_arrays_atomic(project/'plot_data.npz', **coordinates)
 98    paths = render_polynomial_reports(p, rows, coordinates, project)
 99    observations = write_observation_report(project)
100    paths.append(Path(observations['pdf']['path']))
101    summary = {'config': p, 'metrics': rows, 'contract_sha256': digest,
102               'contract': file_record(project/'contract.json'),
103               'table': file_record(project/'summary.csv'), 'plot_data': file_record(project/'plot_data.npz'),
104               'plots': [file_record(path) for path in paths],
105               'reporting_source': file_record(Path(__file__)), 'policies': policy_records,
106               'observation_report': file_record(project/'observation_report.json'),
107               'truth_reference': reference,
108               'coverage_event': 'f inside asymmetric predictor envelope at all 300 saved contexts and every real action',
109               'coverage_claim': 'Empirical conditional coverage; nominal 95% is not a finite-sample guarantee.',
110               'metrics_domain': 'All saved contexts uniformly; independent A~N(0.5,0.45), no truncation.',
111               'width_definition': 'Full transferred width: mean(lower radius + upper radius), after per-context transfer.',
112               'regret_definition': 'J(theta_star)-J(theta_hat), where J is the empirical mean true objective over the saved contexts.',
113               'numerics': 'Exact rational sign certificates for saved floating coefficients; not enclosures of fitting roundoff.'}
114    write_json_atomic(project/'summary.json', summary)
115    return summary

205def _objective_regret(truth, contexts, true_actions, actions):
206    optimal = float(np.mean(truth(contexts, true_actions)))
207    converged = float(np.mean(truth(contexts, actions)))
208    regret = optimal-converged
209    rms = float(np.sqrt(np.mean((actions-.5)**2)))
210    if not np.isclose(regret, .5*rms*rms, atol=1e-12, rtol=1e-10):
211        raise ValueError('Direct objective regret disagrees with the quadratic identity.')
212    return {'objective_at_theta_star': optimal, 'objective_at_theta_hat': converged,
213            'regret': regret, 'rms_action_error': rms}

216def update_regret_report(project):
217    """Replay only direct true-objective regret, its PDF and table from saved policies."""
218    project = Path(project)
219    summary = read_json(project/'summary.json')
220    _verify(summary['table'])
221    _verify(summary['contract'])
222    for plot in summary['plots']:
223        _verify(plot)
224    contract = read_json(project/'contract.json')
225    if contract['sha256'] != summary['contract_sha256'] or contract['contract']['manifest'] != summary['config']:
226        raise ValueError('Report and numerical contract differ.')
227    _verify(contract['environment'])
228    _verify(summary['truth_reference']['arrays'])
229    with np.load(contract['environment']['path'], allow_pickle=False) as saved:
230        contexts = saved['X']
231    with np.load(summary['truth_reference']['arrays']['path'], allow_pickle=False) as saved:
232        true_actions = saved['actions']
233    truth, _ = _models(summary['config'])
234    for row, policy in zip(summary['metrics'], summary['policies']):
235        if row['degree'] != policy['degree'] or policy['contract_sha256'] != summary['contract_sha256']:
236            raise ValueError('Policy and summary provenance differ.')
237        _verify(policy['arrays'])

239            row.update(_objective_regret(truth, contexts, true_actions, saved['actions']))
240    with (project/'summary.csv').open('w', newline='') as stream:

245    summary['plots'] = [file_record(path) if Path(record['path']) == path else record for record in summary['plots']]
246    summary['table'] = file_record(project/'summary.csv')
247    summary['regret_definition'] = 'J(theta_star)-J(theta_hat), where J is the empirical mean true objective over the saved contexts.'
248    summary['regret_reporting_source'] = file_record(Path(__file__))
249    write_json_atomic(project/'summary.json', summary)

258def write_metric_comparison(studies, output_dir):
259    """Replay saved R2, mean transferred radius, regret and coverage as four aligned bar charts.
260
261    Each study specifies project, name, title, and optionally models and selection_note.
262    """
263    import shutil
264    import matplotlib.pyplot as plt
265    from matplotlib.ticker import PercentFormatter
266
267    output_dir = Path(output_dir).resolve()
268    source = Path(__file__).resolve()
269    snapshot = output_dir/'reporting-source'/source.name
270    if snapshot.exists() and snapshot.read_bytes() != source.read_bytes():
271        raise ValueError('Archive the previous metric comparison before changing its source.')
272    request = {'studies': studies, 'output_dir': str(output_dir)}
273    request_path = output_dir/'request.json'
274    if request_path.exists() and read_json(request_path) != request:
275        raise ValueError('Archive the previous metric comparison before changing its selection.')
276
277    points, inputs, groups = [], [], []
278    for study in studies:
279        project = Path(study['project']).resolve()
280        summary = read_json(project/'summary.json')
281        records = [file_record(project/'summary.json'), summary['contract'], summary['table']]
282        contract = read_json(summary['contract']['path'])
283        if contract['sha256'] != summary['contract_sha256']:
284            raise ValueError('Summary and fitted-state contracts differ.')
285        records.append(contract['environment'])
286        records.append(summary['truth_reference']['arrays'])
287        configs = {spec['name']: spec for spec in summary['config'].get('models', [])}
288        selected = study.get('models')
289        group = []
290        for row, policy in zip(summary['metrics'], summary['policies'], strict=True):
291            model = row.get('model', f"degree-{row.get('degree')}")
292            if selected is not None and model not in selected:
293                continue
294            policy_path = project/'policies'/f'{model}.json'
295            if read_json(policy_path) != policy or policy['contract_sha256'] != summary['contract_sha256']:
296                raise ValueError(f'Policy provenance differs for {model}.')
297            records.extend([file_record(policy_path), policy['arrays']])
298            records.extend(attempt['arrays'] for attempt in policy['attempts'])
299            spec = configs.get(model)
300            if spec is None:
301                label = rf'$\phi_{{{row["degree"]}}}$'
302            elif spec['kind'] == 'polynomial':
303                label = rf'$\phi_{{{spec["degree"]},{spec["interaction_order"]}}}$'
304            else:
305                label = f'MLP {spec["hidden_width"]}×{spec["hidden_width"]} ({spec["max_steps"]:,}-step limit)'
306            contexts = row.get('evaluation_contexts', row.get('contexts'))
307            if row['solver_success_contexts'] < contexts:
308                label += ' (policy not fully converged)'
309            radius = (row['mean_lower_radius']+row['mean_upper_radius'])/2
310            if not np.isclose(2*radius, row['mean_full_width'], rtol=1e-12, atol=1e-12):
311                raise ValueError('Asymmetric radii do not match the saved full width.')
312            point = {'study': study['name'], 'model': model, 'label': label,
313                     'headline_repetition': row.get('headline_repetition', 0),
314                     'r_squared': row['r_squared'], 'mean_radius': radius,
315                     'mean_lower_radius': row['mean_lower_radius'],
316                     'mean_upper_radius': row['mean_upper_radius'],
317                     'mean_full_width': row['mean_full_width'], 'regret': row['regret'],
318                     'evaluation_contexts': contexts,
319                     'solver_success_contexts': row['solver_success_contexts'],
320                     'coverage_count': row['transferred_count'],
321                     'coverage_repetitions': row['coverage_repetitions'],
322                     'coverage_probability': row['transferred_coverage'],
323                     'coverage_domain': summary['config']['target']['actions'],
324                     'source_project': str(project)}
325            group.append(point)
326        if not group or (selected is not None and set(selected) != {p['model'] for p in group}):
327            raise ValueError('Selected models are missing from the saved report.')
328        for record in records:
329            _verify(record)
330        inputs.extend(records)
331        points.extend(group)
332        groups.append((study, group))
333
334    output_dir.mkdir(parents=True, exist_ok=True)
335    paths = []
336    style = {'figure.titlesize': 16, 'axes.titlesize': 14, 'axes.labelsize': 12,
337             'xtick.labelsize': 10, 'ytick.labelsize': 10, 'legend.fontsize': 10}
338    radius_label = r'Mean transferred radius $E[(r_l+r_u)/2]$'
339    r2_label = r'True-residual $R^2$ (full Gaussian actions)'
340    regret_label = r'True regret $J(\theta^\star)-J(\widehat\theta)$'
341    panels = [('r_squared', r2_label, r'$R^2$ ↑', '.3f'),
342              ('mean_radius', radius_label, 'Mean radius ↓', '.3f'),
343              ('regret', regret_label, 'Regret ↓', '.4f')]
344    with plt.rc_context(style):
345        for study, group in groups:
346            domain = 'finite action grid' if group[0]['coverage_domain'] == 'finite_grid' else 'all real actions'
347            repetitions = group[0]['coverage_repetitions']
348            study_panels = panels+[('coverage_probability', f'Simultaneous coverage\n({domain}; {repetitions} repetitions)', 'Coverage ↑', '.0%')]
349            fig, axes = plt.subplots(1, 4, figsize=(20, 5.5), sharey=True, constrained_layout=True)
350            fig.suptitle(study['title']+f' — R²/radius/regret: fixed noise; coverage: {repetitions} repetitions')
351            positions = np.arange(len(group))
352            colors = plt.rcParams['axes.prop_cycle'].by_key()['color']
353            for ax, (metric, xlabel, title, fmt) in zip(axes, study_panels):
354                values = [point[metric] for point in group]
355                bars = ax.barh(positions, values, color=[colors[i % len(colors)] for i in positions])
356                ax.bar_label(bars, labels=[format(value, fmt) for value in values], padding=3, fontsize=10)
357                ax.set(xlabel=xlabel, title=title)
358                ax.margins(x=.2)
359                if metric == 'coverage_probability':
360                    ax.set_xlim(0, 1.15)
361                    ax.set_xticks([0, .25, .5, .75, 1])
362                    ax.xaxis.set_major_formatter(PercentFormatter(xmax=1))
363            axes[0].set_yticks(positions, [p['label'] for p in group])
364            axes[0].invert_yaxis()
365            axes[0].set_ylabel('Representation')
366            path = output_dir/f'{study["name"]}_r_squared_radius_regret_coverage_bars.pdf'
367            fig.savefig(path, format='pdf')
368            plt.close(fig)
369            paths.append(path)
370
371    table = output_dir/'metric_comparison.csv'
372    with table.open('w', newline='') as stream:
373        writer = csv.DictWriter(stream, list(points[0]))
374        writer.writeheader()
375        writer.writerows(points)
376    snapshot.parent.mkdir(exist_ok=True)
377    shutil.copyfile(source, snapshot)
378    write_json_atomic(request_path, request)
379    report = {'request': file_record(request_path), 'inputs': inputs,
380              'reporting_source': file_record(source), 'reporting_source_snapshot': file_record(snapshot),
381              'plots': [file_record(path) for path in paths], 'table': file_record(table),
382              'metrics': points,
383              'radius_definition': 'Mean(lower asymmetric radius + upper asymmetric radius)/2; half transferred full width.',
384              'r_squared_definition': 'Known noiseless residual, uniform evaluation contexts and full Gaussian actions.',
385              'regret_definition': 'Saved direct true-objective difference at repository optimizer endpoints.',
386              'coverage_definition': 'Repeated-noise simultaneous transferred coverage in a separate percentage panel; finite-grid for nonlinear.',
387              'comparison_semantics': 'R2/radius/regret use repetition 0; coverage uses all saved repetitions. Within-study comparisons, not matched sample sizes across studies.'}
388    write_json_atomic(output_dir/'provenance.json', report)
389    return report

392def write_observation_report(project):
393    """Replay true/surrogate curves and context-aligned observations with saved noise scales."""
394    from scipy.integrate import quad
395    from scipy.stats import norm
396    import matplotlib
397    matplotlib.use('Agg')
398    import matplotlib.pyplot as plt
399
400    project = Path(project).resolve()
401    contract = read_json(project/'contract.json')
402    repetition = read_json(project/'repetitions/rep-000.json')
403    if repetition['contract_sha256'] != contract['sha256']:
404        raise ValueError('Observation and environment contracts differ.')
405    _verify(contract['environment'])
406    _verify(repetition['arrays'])
407    original_math = next(s for s in contract['contract']['sources']
408                         if s['path'].endswith('/envelopes/polynomial.py'))
409    if original_math['sha256'] != file_record(Path(__file__).with_name('polynomial.py'))['sha256']:
410        raise ValueError('Mathematical query source differs from the saved contract.')
411    with np.load(contract['environment']['path'], allow_pickle=False) as saved:
412        snapshot = dict(saved)
413    with np.load(repetition['arrays']['path'], allow_pickle=False) as saved:
414        state = dict(saved)
415    p = contract['contract']['manifest']
416    truth, residual = _models(p)
417    X, actions, sigma = snapshot['X'], snapshot['actions'], snapshot['sigma']
418    reference = X.mean(axis=0)
419    offsets = (X-reference) @ np.asarray(truth.context)
420    aligned = state['observations']-offsets
421    noise = state['observations']-truth(X, actions)
422    if not np.allclose(noise, sigma*state['innovations'], atol=2e-14, rtol=1e-13):
423        raise ValueError('Saved noise no longer agrees with observation construction.')
424    if not np.allclose(aligned, truth(reference, actions)+noise, atol=2e-14, rtol=1e-13):
425        raise ValueError('Context alignment does not preserve the original noise.')
426    mean, variance = p['design']['action_mean'], p['design']['action_variance']
427    sd = np.sqrt(variance)
428    span = p['reporting']['action_window_standard_deviations']*sd
429    grid = np.linspace(min(mean-span, actions.min()), max(mean+span, actions.max()),
