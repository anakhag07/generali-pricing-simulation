 29def load_polynomial_manifest(path, payload):
 30    """Validate the explicit first-stage scientific contract before sampling."""
 31    from .wild_study import WildEnvelopeManifest
 32    p = payload
 33    if p['study_mode'] != 'polynomial_residual' or p['band_target'] != 'residual':
 34        raise ValueError('Require polynomial_residual mode and residual target.')
 35    name = p['name']
 36    if not name or any(c not in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_' for c in name):
 37        raise ValueError('Require a safe result folder name.')
 38    degrees = p['feature_degrees']
 39    if degrees != sorted(set(degrees)) or any(type(k) is not int or k < 1 or k > 6 for k in degrees):
 40        raise ValueError('Require unique increasing polynomial degrees in 1..6.')
 41    if type(p['n']) is not int or p['n'] <= max(degrees)+4:
 42        raise ValueError('Require n greater than every feature dimension.')
 43    if p['design']['context_dim'] != 3 or p['design']['action_variance'] <= 0:
 44        raise ValueError('Require three contexts and positive action variance.')
 45    if 'models' in p:
 46        from .structured_residual import validate_structured
 47        validate_structured(p)
 48    elif p['target'] != {'contexts': 'calibration', 'actions': 'real_line'}:
 49        raise ValueError('Require all calibration contexts and every real action.')
 50    if p['bootstrap']['multipliers'] != 'rademacher' or p['bootstrap']['residuals'] != 'raw':
 51        raise ValueError('Require raw-residual Rademacher bootstrap.')
 52    if p['bootstrap']['calibration'] != 'coefficient_ellipsoid' or p['transfer'] != 'asymmetric':
 53        raise ValueError('Require coefficient ellipsoid and asymmetric transfer.')
 54    if type(p['bootstrap']['B']) is not int or p['bootstrap']['B'] < 2 or not 0 < p['bootstrap']['delta'] < 1:
 55        raise ValueError('Invalid bootstrap count or miscoverage level.')
 56    if p['noise']['type'] != 'squared_exponential_gp_restriction':
 57        raise ValueError('Require the exact squared-exponential finite restriction.')
 58    if not 0 < p['noise']['sigma_min'] < p['noise']['sigma_max'] or min(p['noise']['lengthscale_x'], p['noise']['lengthscale_a']) <= 0:
 59        raise ValueError('Invalid noise scale/lengthscale.')
 60    for key in ('truth', 'residual'):
 61        model = p[key]
 62        if len(model['context']) != 3 or not model['action'] or not np.isfinite([*model['context'], *model['action']]).all():
 63            raise ValueError('Require explicit finite additive function coefficients.')
 64    for key in ('repetitions', 'repetitions_per_task'):
 65        if type(p[key]) is not int or p[key] < 1:
 66            raise ValueError('Require positive repetition counts.')
 67    for key in ('contexts', 'actions', 'profile', 'observations', 'method', 'optimizer'):
 68        if type(p['seeds'][key]) is not int or p['seeds'][key] < 0:
 69            raise ValueError('Require separate named nonnegative integer seed roots.')
 70    o = p['optimizer']
 71    if o['method'] != 'stein_difference' or o['perturbation_space'] != 'u':
 72        raise ValueError('Require native action-space Stein differences.')
 73    if o['step_rule'] not in ('constant', 'armijo', 'l-bfgs-b') or not o['starts']:
 74        raise ValueError('Invalid repository step rule or starts.')
 75    if min(o['t_steps'], o['n_grad_samples'], o['smoothing_sd'], o['step_size'], o['gradient_tolerance']) <= 0:
 76        raise ValueError('Require positive optimizer budget and scales.')
 77    launch = p['launch']
 78    return WildEnvelopeManifest(p, Path(path).resolve(), SimpleNamespace(
 79        mode=launch['mode'], array=launch['array'], array_max_parallel=launch['array_max_parallel']))

 82def prepare_environment(p):
 83    """Draw the fixed design and exact finite GP restriction using separate streams."""
 84    n, seeds, design, noise = p['n'], p['seeds'], p['design'], p['noise']
 85    X = np.random.default_rng(seeds['contexts']).normal(size=(n, 3))
 86    action_z = np.random.default_rng(seeds['actions']).normal(size=n)
 87    actions = design['action_mean']+np.sqrt(design['action_variance'])*action_z
 88    distance = np.sum((X[:, None, :]-X[None, :, :])**2, axis=2)/noise['lengthscale_x']**2
 89    distance += (actions[:, None]-actions[None, :])**2/noise['lengthscale_a']**2
 90    covariance = np.exp(-distance/2)
 91    cholesky = np.linalg.cholesky(covariance)  # No silent nugget or kernel replacement.
 92    gp_z = np.random.default_rng(seeds['profile']).normal(size=n)
 93    profile = cholesky @ gp_z
 94    sigma = noise['sigma_min']+(noise['sigma_max']-noise['sigma_min'])*expit(noise['amplitude']*profile)
 95    return dict(X=X, actions=actions, action_innovations=action_z, gp_covariance=covariance,
 96                gp_cholesky=cholesky, gp_innovations=gp_z, gp_values=profile, sigma=sigma)

111def prepare(manifest, runs_root):
112    """Create or verify immutable environment, source snapshot and experiment contract."""
113    p = manifest.payload
114    project = Path(runs_root)/manifest.name
115    project.mkdir(parents=True, exist_ok=True)
116    root, sources = _runtime_sources()
117    payload = {'manifest': p, 'sources': [file_record(s) for s in sources],
118               'packages': {name: version(name) for name in ('numpy', 'scipy', 'sympy')}}
119    digest = hashlib.sha256(json.dumps(payload, sort_keys=True).encode()).hexdigest()
120    path = project/'contract.json'
121    if path.exists():
122        saved = read_json(path)
123        if saved['sha256'] != digest or saved['environment'] != file_record(project/'inputs.npz'):
124            raise ValueError('Completed input/source contract differs; use a new result folder.')
125        with np.load(project/'inputs.npz', allow_pickle=False) as data:
126            snapshot = dict(data)
127    else:
128        snapshot = prepare_environment(p)
129        save_arrays_atomic(project/'inputs.npz', **snapshot)
130        copies = []
131        for source in sources:
132            target = project/'source'/source.relative_to(root)
133            target.parent.mkdir(parents=True, exist_ok=True)
134            shutil.copyfile(source, target)
135            copies.append(file_record(target))
136        write_json_atomic(project/'config.json', p)
137        saved = {'sha256': digest, 'contract': payload, 'environment': file_record(project/'inputs.npz'),
138                 'source_snapshot': copies, 'source_worktree': str(root),
139                 'repository_head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip()}
140        write_json_atomic(path, saved)
141        guide = 'synthetic_residual_extensions.md' if 'models' in p else 'synthetic_polynomial_residual.md'
142        if p.get('reporting', {}).get('fit_diagnostics'):
143            guide = 'synthetic_mlp_capacity.md'
144        shutil.copyfile(root/'docs'/guide, project/'EXPERIMENT.md')
145    designs = {} if 'models' in p else {k: WildFeatureDesign(polynomial_features(snapshot['X'], snapshot['actions'], k))
146                                       for k in p['feature_degrees']}
147    return project, digest, snapshot, designs

150def run_repetition(p, index, project, digest, snapshot, designs):
151    """Fit paired feature degrees and save actual all-real coverage decisions."""
152    stem = project/'repetitions'/f'rep-{index:03d}'
153    if stem.with_suffix('.json').exists():
154        record = read_json(stem.with_suffix('.json'))
155        if record['contract_sha256'] != digest or record['arrays'] != file_record(stem.with_suffix('.npz')):
156            raise ValueError('Corrupt or incompatible saved repetition.')
157        return record
158    seeds = {key: derive_seed(p['seeds'][key], f'repetition:{index}') for key in ('observations', 'method')}
159    innovations = np.random.default_rng(seeds['observations']).normal(size=p['n'])
160    signs = 2*np.random.default_rng(seeds['method']).integers(0, 2, size=(p['bootstrap']['B'], p['n']), dtype=np.int8)-1
161    truth, residual = _models(p)
162    X, actions = snapshot['X'], snapshot['actions']
163    observations = truth(X, actions)+snapshot['sigma']*innovations
164    responses = observations-(truth(X, actions)-residual(X, actions))
165    arrays = {'innovations': innovations, 'multipliers': signs, 'observations': observations,
166              'residual_observations': responses}
167    events, diagnostics = {}, {}
168    if 'models' in p:
169        from .structured_residual import fit_structured_models
170        events, diagnostics = fit_structured_models(p, index, project, digest, snapshot, responses, signs, arrays)
171    for degree in ([] if 'models' in p else p['feature_degrees']):
172        band, state = fit_polynomial_band(X, actions, responses, degree, signs,
173                                         p['bootstrap']['delta'], design=designs[degree])
174        event = coverage_events(band, residual)
175        events[str(degree)] = event
176        arrays.update({f'degree{degree}_{key}': value for key, value in state.items()})
177    save_arrays_atomic(stem.with_suffix('.npz'), **arrays)
178    record = {'index': index, 'contract_sha256': digest, 'seeds': seeds,
179              'multiplier_sha256': array_sha256(signs), 'events': events,
180              'arrays': file_record(stem.with_suffix('.npz'))}
181    if 'models' in p:
182        record['model_diagnostics'] = diagnostics
183    write_json_atomic(stem.with_suffix('.json'), record)
184    print(f"Repetition {index+1}/{p['repetitions']}: transferred "
185          f"{[int(event['transferred']) for event in events.values()]}", flush=True)
186    return record

195class ContextActions(Policy):
196    """One free action per context in the fixed ordered full batch."""
197
198    def value(self, theta, x_batch):
199        if np.shape(theta) != (len(x_batch),):
200            raise ValueError('Identity policy requires the full ordered context batch.')
201        return np.asarray(theta)
202
203    def weighted_grad(self, theta, x_batch, weights):
204        self.value(theta, x_batch)
205        return np.asarray(weights)

208class PolynomialLCBObjective(Objective):
209    """Negative asymmetric LCB exposed only through exact function-value queries."""
210
211    policy = ContextActions()
212
213    def __init__(self, band, truth, residual):
214        self.band, self.truth, self.residual = band, truth, residual
215        self.query_count = 0
216        self.query_min = np.full(len(band.contexts), np.inf)
217        self.query_max = np.full(len(band.contexts), -np.inf)
218
219    def _value_batch_many(self, x_batch, actions):
220        if not np.array_equal(x_batch, self.band.contexts):
221            raise ValueError('Full ordered context queries are required.')
222        a = np.asarray(actions)
223        self.query_count += a.size
224        self.query_min = np.minimum(self.query_min, a.min(axis=0))
225        self.query_max = np.maximum(self.query_max, a.max(axis=0))
226        _, _, lower, _ = self.band.evaluate(a)
227        return -self.truth(x_batch, a)+self.residual(x_batch, a)+lower
228
229    def _value_batch(self, x_batch, actions):
230        return self._value_batch_many(x_batch, np.asarray(actions)[None, :])[0]
231
232    def value(self, theta, x_batch):
233        return float(np.mean(self._value_batch(x_batch, theta)))
234
235    def grad(self, theta, x_batch):
236        raise AssertionError('The LCB must use Stein function-value queries.')
237

239class _TruthObjective(Objective):
240    policy = ContextActions()
241
242    def __init__(self, truth):
243        self.truth = truth
244
245    def value(self, theta, x_batch):
246        return -float(np.mean(self.truth(x_batch, theta)))
247
248    def grad(self, theta, x_batch):
249        return -np.polynomial.polynomial.polyval(theta, np.polynomial.polynomial.polyder(self.truth.action))/len(x_batch)

252def optimize_headline(p, project, digest, snapshot):
253    """Run/replay repository truth and personalized Stein endpoints for repetition zero."""
254    output = project/'policies'
255    output.mkdir(exist_ok=True)
256    truth, residual = _models(p)
257    if 'models' in p:
258        from .structured_residual import evaluation_rows, restore_structured_band
259        X = snapshot['X'][evaluation_rows(p)]
260        entries = [(m['name'], m['name']) for m in p['models']]
261    else:
262        X = snapshot['X']
263        entries = [(k, f'degree-{k}') for k in p['feature_degrees']]
264    n, settings = len(X), p['optimizer']
265    reference_path = output/'truth.json'
266    if reference_path.exists():
267        reference = _verified_record(reference_path, digest)
268    else:
269        theta, trace = run_first_order_minimize(np.zeros(n), X, _TruthObjective(truth),
270            t_steps=100, n_grad_samples=1, sigma=0., grad_norm_tol=1e-12, ftol=1e-14)
271        if not trace.optimizer_success or not np.allclose(theta, .5, atol=1e-8, rtol=0):
272            raise RuntimeError('Repository truth solve failed the supplied quadratic reference check.')
273        save_arrays_atomic(output/'truth.npz', actions=theta, theta_values=np.asarray(trace.theta_values),
274                           losses=trace.objective_values)
275        reference = {'contract_sha256': digest, 'solver': 'optimization.solvers.run_first_order_minimize',
276                     'success': bool(trace.optimizer_success), 'max_error_from_supplied_truth': float(np.max(abs(theta-.5))),
277                     'arrays': file_record(output/'truth.npz')}
278        write_json_atomic(reference_path, reference)
279    with np.load(project/'repetitions/rep-000.npz', allow_pickle=False) as saved:
280        state = dict(saved)
281    for degree, label in entries:
282        path = output/f'{label}.json'
283        if path.exists():
284            _verified_record(path, digest)
285            continue
286        band = (restore_structured_band(p, snapshot, state, degree) if 'models' in p
287                else restore_band(snapshot, state, degree))
288        attempts, vectors, losses = [], [], []
289        for j, start in enumerate(settings['starts']):
290            attempt_path = output/f'{label}-start-{j}.json'
291            if attempt_path.exists():
292                info = _verified_record(attempt_path, digest)
293                with np.load(info['arrays']['path'], allow_pickle=False) as saved:
294                    theta, final_losses = saved['actions'], saved['final_losses']
295            else:
296                seed = derive_seed(p['seeds']['optimizer'], f'stein-difference:start:{j}')
297                batch_seed = derive_seed(p['seeds']['optimizer'], f'batch:start:{j}')
298                objective = PolynomialLCBObjective(band, truth, residual)
299                theta, trace = run_stein_difference_minimize(np.full(n, start), X, objective,
300                    np.random.default_rng(seed), t_steps=settings['t_steps'],
301                    n_grad_samples=settings['n_grad_samples'], sigma=settings['smoothing_sd'],
302                    perturbation_space='u', algorithm=settings['step_rule'],
303                    step_size=n*settings['step_size'], grad_norm_tol=settings['gradient_tolerance']/np.sqrt(n),
304                    ftol=settings.get('ftol'),
305                    gradient_rng=np.random.default_rng(seed), batch_rng=np.random.default_rng(batch_seed))
306                final_losses = objective._value_batch(X, theta)
307                if not np.isfinite(theta).all() or not np.isfinite(final_losses).all():
308                    raise FloatingPointError('Nonfinite repository endpoint; headline remains incomplete.')
309                arrays_path = attempt_path.with_suffix('.npz')
310                save_arrays_atomic(arrays_path, actions=theta, theta_values=np.asarray(trace.theta_values),
311                    losses=trace.objective_values, gradient_norms=trace.theta_grad_norms,
312                    final_losses=final_losses, query_min=objective.query_min, query_max=objective.query_max)
313                info = {'contract_sha256': digest, 'start': start, 'seed': seed, 'batch_seed': batch_seed,
314                        'success': bool(trace.optimizer_success), 'message': trace.optimizer_message,
315                        'steps': int(trace.steps[-1]), 'query_count': objective.query_count,
316                        'arrays': file_record(arrays_path)}
317                write_json_atomic(attempt_path, info)
318            attempts.append(info)
319            vectors.append(theta)
320            losses.append(final_losses)
321            print(f"Degree {degree}, start {start}: success={info['success']}, steps={info['steps']}", flush=True)
322        # Only actual repository endpoints compete; no action/evaluation grid is searched.
323        selected = np.argmin(np.asarray(losses), axis=0)
324        actions = np.asarray(vectors)[selected, np.arange(n)]
325        polynomial_certificate = p['target']['actions'] == 'real_line'
326        certificates = (certify_policy_gap(band, truth, residual, actions, settings['global_gap_tolerance'])
327                        if polynomial_certificate else np.full(n, -1, dtype=np.int8))
328        selected_success = np.array([attempts[int(j)]['success'] for j in selected])
329        save_arrays_atomic(path.with_suffix('.npz'), actions=actions, selected_start=selected,
330                           all_start_actions=np.asarray(vectors), gap_certified=certificates,
331                           selected_solver_success=selected_success)
332        record = {'contract_sha256': digest, 'degree': degree,
333                  'solver': 'optimization.solvers.run_stein_difference_minimize', 'attempts': attempts,
334                  'selected_success_count': int(selected_success.sum()),
335                  'global_gap_tolerance': settings['global_gap_tolerance'],
336                  'global_gap_certified_count': int(certificates.sum()) if polynomial_certificate else None,
337                  'arrays': file_record(path.with_suffix('.npz'))}
338        write_json_atomic(path, record)
339        print(f"Model {degree}: global gap certificates {record['global_gap_certified_count']}/{n}", flush=True)

354def build_polynomial_launch_plan(manifest, *, runs_root=None, force=False):
355    """Reuse the family launcher, preserving completed repetitions and solver endpoints."""
356    if force:
357        raise ValueError('Polynomial runs preserve completed artifacts; choose a fresh folder for revisions.')
358    p = manifest.payload
359    count = int(np.ceil(p['repetitions']/p['repetitions_per_task']))
360    def run_task(index, context):
361        if not 0 <= index < count:
362            raise IndexError(index)
363        project, digest, snapshot, designs = prepare(manifest, context.runs_root)
364        indices = range(index*p['repetitions_per_task'], min((index+1)*p['repetitions_per_task'], p['repetitions']))
365        for repetition in indices:
366            run_repetition(p, repetition, project, digest, snapshot, designs)
367            if repetition == 0:
368                optimize_headline(p, project, digest, snapshot)
369        return {'repetitions': list(indices)}
370    def collect(context):
371        if 'models' in p:
372            from .structured_reporting import collect_structured
373            return collect_structured(Path(context.runs_root)/manifest.name)
374        from .polynomial_reporting import collect_polynomial
375        return collect_polynomial(Path(context.runs_root)/manifest.name)
376    def run_all(context):
377        for index in range(count):
378            run_task(index, context)
379        return collect(context)
380    return LaunchPlan(name=manifest.name, task_count=count, requires_jax=False,
381                      run_task=run_task, run_all=run_all, collect=collect, runs_root=runs_root,
382                      default_launch=manifest.launch.mode, default_array=manifest.launch.array == 'repetition',
383                      slurm_profile=replace(CPU_PROFILE, cpus_per_task=1, memory='4G'))
