137class SoftmaxPolicy(Policy):
138    """Softmax policy: $$u = 0.5 + \\sigma(\\theta^\\top \\phi(x)) \\in (0.5, 1.5)$$."""
139
140    kind: str = policy_softmax
141
142    def value(self, theta: np.ndarray, x: StateVector) -> float:
143        features = _phi(x)
144        theta_arr = np.asarray(theta, dtype=float)
145        if theta_arr.size < features.size:
146            raise ValueError("theta must have at least state_dim + 1 elements for softmax policy.")
147        z = float(np.dot(theta_arr[: features.size], features))
148        sigma = float(_sigmoid(np.asarray([z]))[0])
149        return float(0.5 + sigma)
150
151    def grad(self, theta: np.ndarray, x: StateVector) -> np.ndarray:
152        features = _phi(x)
153        theta_arr = np.asarray(theta, dtype=float)
154        if theta_arr.size < features.size:
155            raise ValueError("theta must have at least state_dim + 1 elements for softmax policy.")
156        z = float(np.dot(theta_arr[: features.size], features))
157        sigma = float(_sigmoid(np.asarray([z]))[0])
158        du_dz = sigma * (1.0 - sigma)
159        grad = np.zeros_like(theta_arr)
160        grad[: features.size] = du_dz * features
161        return grad
162
163    def value_batch(self, theta: np.ndarray, x_array: np.ndarray) -> np.ndarray:
164        x_arr = np.asarray(x_array, dtype=float)
165        if x_arr.ndim != 2:
166            raise ValueError("x_array must be a 2D array.")
167        features = _phi_batch(x_arr)
168        theta_arr = np.asarray(theta, dtype=float)
169        if theta_arr.size < features.shape[1]:
170            raise ValueError("theta must have at least state_dim + 1 elements for softmax policy.")
171        z = features @ theta_arr[: features.shape[1]]
172        return (0.5 + _sigmoid(z)).astype(float)
173
174    def grad_batch(self, theta: np.ndarray, x_array: np.ndarray) -> np.ndarray:
175        x_arr = np.asarray(x_array, dtype=float)
176        if x_arr.ndim != 2:
177            raise ValueError("x_array must be a 2D array.")
178        features = _phi_batch(x_arr)
179        theta_arr = np.asarray(theta, dtype=float)
180        if theta_arr.size < features.shape[1]:
181            raise ValueError("theta must have at least state_dim + 1 elements for softmax policy.")
182        z = features @ theta_arr[: features.shape[1]]
183        sigma = _sigmoid(z)
184        du_dz = sigma * (1.0 - sigma)
185        n_samples = x_arr.shape[0]
186        grad = np.zeros((n_samples, theta_arr.size), dtype=float)
187        grad[:, : features.shape[1]] = du_dz[:, None] * features
188        return grad

 41    Computes theta-gradients via chain rule through the attached policy.
 42    """

 46    beta_2: float
 47    beta_3: np.ndarray
 48    beta_4: float

 50    def __post_init__(self) -> None:
 51        beta_1 = np.asarray(self.beta_1, dtype=float)
 52        beta_2 = float(self.beta_2)
 53        beta_3 = np.asarray(self.beta_3, dtype=float)
 54        beta_4 = float(self.beta_4)
