469    """Plot objective value as a function of u.
470
471    Uses the objective's value_at_u method for computing values at fixed u.
472    """
473    trace_items = _ordered_traces(traces)
474    if not trace_items:
475        return
476    path = _ensure_plot_dir(plot_dir)
477    x_arr = np.asarray(x_samples, dtype=float)
478    if x_arr.ndim != 2:
479        raise ValueError("x_samples must be a 2D array.")
480
481    u_values: list[float] = []
482    for _, trace in trace_items:
483        u_values.extend(list(trace.u_values))
484    if u_star is not None:
485        u_values.append(float(u_star))
486    if u_values:
487        u_min = float(min(u_values))
488        u_max = float(max(u_values))
489        if np.isclose(u_min, u_max):
490            pad = 0.1 if u_min == 0.0 else abs(u_min) * 0.1
491        else:
492            pad = 0.1 * (u_max - u_min)
493        u_grid = np.linspace(u_min - pad, u_max + pad, 200)
494    else:
495        u_grid = np.linspace(-0.5, 0.5, 200)
496
497    # Use value_at_u method if available
498    value_at_u_fn = getattr(objective, "value_at_u", None)
499    if not callable(value_at_u_fn):
500        return
501    value_at_u_typed = cast(Callable[[np.ndarray, float], float], value_at_u_fn)
502
503    def value_at_u_scalar(u: float) -> float:
504        return float(value_at_u_typed(x_arr, float(u)))
505
506    obj_grid = [value_at_u_scalar(float(u)) for u in u_grid]
507
508    fig, ax = plt.subplots(1, 1, figsize=(8, 5))
509
510    ax.plot(u_grid, obj_grid, color="black", label="objective", alpha=_LINE_ALPHA, linewidth=_LINE_WIDTH)
511    for name, trace in trace_items:
512        style = ESTIMATOR_STYLES[name]
513        zorder = 4 if name == "gauss_stein" else 3
514        ax.scatter(
515            trace.u_values,
516            trace.objective_values,
517            color=style["color"],
518            label=style["label"],
519            marker=style["marker"],
520            edgecolors=style["color"],
521            linewidths=0.4,
522            alpha=0.75,
523            zorder=zorder,
524        )
525    ax.set_ylabel("Objective value")
526    ax.set_xlabel("u")
527    if u_star is not None:
528        ax.axvline(
529            u_star,
530            color="#444444",
531            linestyle="--",
532            linewidth=1.2,
533            alpha=0.8,
534            label="u*",
535        )
536    ax.legend()
537    ax.grid(True, alpha=0.3)
538
539    fig.tight_layout()
540    fig.savefig(path / "objective_u_slice.png", dpi=200)
541    plt.close(fig)
